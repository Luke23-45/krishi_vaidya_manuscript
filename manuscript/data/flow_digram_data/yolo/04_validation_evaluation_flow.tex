\begin{tikzpicture}[

  flow/.style={-{Latex[length=2.8mm,width=2mm]}, line width=0.8pt,
               draw=pri!70, rounded corners=6pt},
  flowret/.style={-{Latex[length=2.2mm,width=1.5mm]}, line width=0.5pt,
                  draw=muted!45, densely dashed},
  card/.style={rounded corners=10pt, draw=ink!10, line width=0.55pt,
               fill=white, align=left, inner sep=8pt,
               font=\sffamily\scriptsize, text=ink,
               blur shadow={shadow blur steps=5, shadow xshift=0.6pt,
               shadow yshift=-1pt, shadow opacity=13}},
  groupbox/.style={rounded corners=11pt, draw=#1!22, line width=0.4pt,
                   fill=#1!3},

                   rounded corners=2pt},
  titlebar/.style={rounded corners=8pt, fill=pri, text=white,
                   inner xsep=14pt, inner ysep=7pt,
                   font=\sffamily\bfseries\large},
  procnode/.style={card, align=center, inner sep=7pt,
                   minimum width=3.8cm, minimum height=1.1cm,
                   fill=#1!4, draw=#1!25},
  arrlbl/.style={font=\sffamily\tiny\itshape, text=muted,
                 fill=white, inner sep=1.5pt, rounded corners=1pt},
  note/.style={rounded corners=7pt, draw=ink!10, fill=ink!2,
               inner sep=8pt, align=left, font=\sffamily\tiny,
               text=ink!78, text width=14.2cm}
]

\definecolor{pri}{HTML}{16A34A}   % Green
\definecolor{sec}{HTML}{0284C7}   % Light Blue
\definecolor{ter}{HTML}{D97706}   % Amber
\definecolor{quat}{HTML}{7C3AED}  % Purple
\definecolor{bg}{HTML}{F8FAFC}
\definecolor{ink}{HTML}{0F172A}
\definecolor{muted}{HTML}{64748B}

% ━━ TITLE ━━
\node[titlebar, minimum width=15.2cm] at (0,0.3)
  {YOLO Validation \& Evaluation Flow};
\node[font=\sffamily\footnotesize, text=muted, text width=14cm,
      align=center] at (0,-1.0)
  {Evaluating the fine-tuned model against the test set, computing
   NMS and bounding box metrics, and generating visual reports.};

% ━━ ROW 1: INPUT ━━
\node[procnode=pri] (testset) at (0,-2.3) {
  \textbf{Test Set}\\[-1pt]
  {\tiny 10\% holdout}};

% ━━ ROW 2: FORK ━━
\node[procnode=sec] (infer) at (-3.8,-4.3) {
  \textbf{Inference}\\[2pt]
    \textcolor{ink!72}{\tiny Forward Pass}\\[-1pt]
  \textcolor{ink!72}{{\tiny NMS (Non-Max Suppression)}}};
\node[procnode=ter] (metric) at (3.8,-4.3) {
  \textbf{Metrics Engine}\\[2pt]
    \textcolor{ink!72}{\tiny mAP@50, mAP@50-95}\\[-1pt]
  \textcolor{ink!72}{{\tiny Precision, Recall}}};
\draw[flow] (testset.south west) -- ++(0,-0.3) -| (infer.north);
\draw[flow] (testset.south east) -- ++(0,-0.3) -| (metric.north) node[arrlbl, pos=0.8, right] {truth labels};

\draw[flow] (infer.east) -- (metric.west) node[arrlbl, midway, above] {predictions};

% ━━ ROW 3: OUTPUTS ━━
\node[procnode=quat] (curves) at (-4,-6.8) {
  \textbf{Performance Curves}\\[-1pt]
  {\tiny F1-Score, PR Curve}};

\node[procnode=quat] (cm) at (0,-6.8) {
  \textbf{Confusion Matrix}\\[-1pt]
  {\tiny Class accuracy}};

\node[procnode=sec] (overlay) at (4,-6.8) {
  \textbf{Visual Overlays}\\[-1pt]
  {\tiny BBoxes drawn on images}};

\draw[flow] (metric.south west) -- ++(0,-0.4) -| (curves.north);
\draw[flow] (metric.south) -- ++(0,-0.4) -| (cm.north);
\draw[flow] (infer.south) -- ++(0,-0.4) -| (overlay.north) node[arrlbl, pos=0.6, above] {renders};

% ━━ NOTE ━━
\node[note] at (0,-8.8) {
  \textbf{Evaluation}\;\textcolor{muted}{\textemdash}\;
  Non-Maximum Suppression (NMS) filters overlapping bounding boxes before
  predictions are compared to ground truth. Key metrics like mAP (Mean Average
  Precision) at various IoU thresholds provide a holistic view of the model's
  localization and classification accuracy.};

% ━━ BACKGROUND ━━
\begin{scope}[on background layer]
  \node[rounded corners=14pt, fill=bg!45, draw=ink!4,
        line width=0.35pt, inner sep=14pt,
        fit=(testset)(infer)(metric)(curves)(overlay)] {};

  \node[groupbox=sec, fit=(infer)(metric),
        inner xsep=10pt, inner ysep=8pt] (gEval) {};
  \node[grouplbl=sec, anchor=north west]
        at ([xshift=2pt]gEval.north west) {Evaluation Loop};
\end{scope}

\end{tikzpicture}%
