% ━━ TITLE ━━
\node[titlebar, minimum width=15.2cm] at (0,0.3)
  {YOLO Training Pipeline Flow};
\node[font=\sffamily\footnotesize, text=muted, text width=14cm,
      align=center] at (0,-1.0)
  {Ultralytics engine fine-tuning YOLO11n on the custom agricultural dataset,
   with integrated logging and automated model checkpointing.};

% ━━ ROW 1: INPUTS ━━
\node[procnode=sec] (dataset) at (-3.5,-2.4) {
  \textbf{Custom Dataset}\\[-1pt]
  {\tiny \texttt{dataset.yaml}}};

\node[procnode=ter] (basemodel) at (3.5,-2.4) {
  \textbf{Base Model}\\[-1pt]
  {\tiny \texttt{yolo11n.pt} (COCO init)}};

% ━━ ROW 2: ENGINE ━━
\node[widenode=pri] (train) at (0,-4.6) {
  \textbf{Ultralytics Training Engine}\\[2pt]
  \textcolor{ink!72}{\tiny $\bullet$ Optimizer: AdamW, LR: $0.001$, Cosine Annealing}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ Loss: CIoU (Box), BCE (Class), DFL}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ Epochs: 100, Batch Size: 32, Image Size: 640}};

\draw[flow] (dataset.south) -- ++(0,-0.4) -| ([xshift=-15pt]train.north);
\draw[flow] (basemodel.south) -- ++(0,-0.4) -| ([xshift=15pt]train.north);

% ━━ ROW 3: OUTPUTS & LOGGING ━━
\node[procnode=quat] (logs) at (-4,-7.2) {
  \textbf{Experiment Tracking}\\[-1pt]
  {\tiny Weights \& Biases (W\&B)}\\[-1pt]
  {\tiny TensorBoard}};

\node[procnode=pri, fill=pri!8, draw=pri!35] (best) at (0,-7.2) {
  \textbf{Best Checkpoint}\\[-1pt]
  {\tiny \texttt{best.pt} (highest mAP)}};

\node[procnode=muted] (last) at (4,-7.2) {
  \textbf{Last Checkpoint}\\[-1pt]
  {\tiny \texttt{last.pt} (resume)}};

\draw[flow] (train.south west) -- ++(0,-0.4) -| (logs.north);
\draw[flow] (train.south) -- (best.north);
\draw[flow] (train.south east) -- ++(0,-0.4) -| (last.north);

% ━━ NOTE ━━
\node[note] at (0,-9.2) {
  \textbf{Training Dynamics}\;\textcolor{muted}{\textemdash}\;
  Transfer learning from COCO pre-trained weights significantly accelerates convergence.
  The engine tracks mAP@0.5:0.95 on the validation set at every epoch, automatically
  saving \texttt{best.pt} to prevent overfitting.};
