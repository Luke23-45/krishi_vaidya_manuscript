\resizebox{0.98\textwidth}{!}{%
\begin{tikzpicture}[
  font=\sffamily,
  flow/.style={-{Latex[length=2.8mm,width=2mm]}, line width=0.8pt,
               draw=pri!70, rounded corners=6pt},
  flowret/.style={-{Latex[length=2.2mm,width=1.5mm]}, line width=0.5pt,
                  draw=muted!45, densely dashed},
  card/.style={rounded corners=10pt, draw=ink!10, line width=0.55pt,
               fill=white, align=left, inner sep=8pt,
               font=\sffamily\scriptsize, text=ink,
               blur shadow={shadow blur steps=5, shadow xshift=0.6pt,
               shadow yshift=-1pt, shadow opacity=13}},
  groupbox/.style={rounded corners=11pt, draw=#1!22, line width=0.4pt,
                   fill=#1!3},
  grouplbl/.style={font=\sffamily\bfseries\tiny, text=#1!70,
                   fill=white, inner xsep=3pt, inner ysep=1.5pt,
                   rounded corners=2pt},
  titlebar/.style={rounded corners=8pt, fill=pri, text=white,
                   inner xsep=14pt, inner ysep=7pt,
                   font=\sffamily\bfseries\large},
  procnode/.style={card, align=center, inner sep=6pt,
                   minimum width=3.4cm, minimum height=1.1cm,
                   fill=#1!4, draw=#1!25},
  badge/.style={circle, fill=#1, text=white, font=\sffamily\bfseries\tiny,
                inner sep=1.5pt, minimum size=14pt},
  arrlbl/.style={font=\sffamily\tiny\itshape, text=muted,
                 fill=white, inner sep=1.5pt, rounded corners=1pt},
  note/.style={rounded corners=7pt, draw=ink!10, fill=ink!2,
               inner sep=8pt, align=left, font=\sffamily\tiny,
               text=ink!78, text width=14.2cm}
]

\definecolor{pri}{HTML}{16A34A}   % Green
\definecolor{sec}{HTML}{0284C7}   % Light Blue
\definecolor{ter}{HTML}{D97706}   % Amber
\definecolor{quat}{HTML}{7C3AED}  % Purple
\definecolor{bg}{HTML}{F8FAFC}
\definecolor{ink}{HTML}{0F172A}
\definecolor{muted}{HTML}{64748B}

% ━━ TITLE ━━
\node[titlebar, minimum width=15.2cm] at (0,0.3)
  {YOLO Data Pipeline Flow};
\node[font=\sffamily\footnotesize, text=muted, text width=14cm,
      align=center] at (0,-1.0)
  {Conversion of annotations to YOLO format, train/val/test splitting,
   and online/offline image augmentations for robust object detection.};

% ━━ ROW 1: INGESTION & CONVERSION ━━
\node[procnode=sec] (raw) at (-5.0,-2.5) {
  \textbf{Raw Annotations}\\[-1pt]
  {\tiny JSON/XML bounding boxes}};

\node[procnode=pri] (convert) at (0,-2.5) {
  \textbf{Format Conversion}\\[-1pt]
  {\tiny normalized $(x_c, y_c, w, h)$}};

\node[procnode=pri] (split) at (5.0,-2.5) {
  \textbf{Dataset Splitting}\\[-1pt]
  {\tiny 80\% Train, 10\% Val, 10\% Test}};

\draw[flow] (raw) -- (convert);
\draw[flow] (convert) -- (split);

% ━━ ROW 2: AUGMENTATION STRATEGIES ━━
\node[procnode=ter] (offline) at (2.0,-5.0) {
  \textbf{Offline Augmentation}\\[-1pt]
  {\tiny Rotation, Flipping, Contrast}};

\node[procnode=quat] (online) at (-2.0,-5.0) {
  \textbf{Online Augmentation}\\[-1pt]
  {\tiny Mosaic, MixUp (YOLO native)}};

\draw[flow] (split.south) |- (offline.east);

% ━━ ROW 3: FINAL OUTPUT ━━
\node[procnode=sec, fill=sec!8, draw=sec!35, minimum width=4cm] (final) at (0,-7.5) {
  \textbf{YOLO Training Structure}\\[2pt]
    \textcolor{ink!72}{\tiny $\bullet$ /images/train, /labels/train}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ /images/val, /labels/val}\\[-1pt]
  \textcolor{ink!72}{{\tiny $\bullet$ dataset.yaml}}};
\draw[flow] (offline.west) -- (online.east) node[arrlbl, midway, above] {cache};
\draw[flow] (online.south) |- (final.west);

% Alternative path straight from split to final (val/test sets)
\draw[flowret] (split.south east) -- ++(0.5,-0.5) |- (final.east)
  node[arrlbl, pos=0.8, above] {val / test (no aug)};

% ━━ NOTE ━━
\node[note] at (0,-9.5) {
  \textbf{Augmentation Strategy}\;\textcolor{muted}{\textemdash}\;
  Offline augmentation expands the static dataset to combat class imbalance.
  Online augmentation (Mosaic, MixUp) is applied dynamically during YOLO training
  by the dataloader to improve model generalization and detection robustness on
  variable scales.};

% ━━ BACKGROUND ━━
\begin{scope}[on background layer]
  \node[rounded corners=14pt, fill=bg!45, draw=ink!4,
        line width=0.35pt, inner sep=14pt,
        fit=(raw)(split)(offline)(online)(final)] {};

  \node[groupbox=pri, fit=(convert)(split),
        inner xsep=8pt, inner ysep=8pt] (gPrep) {};
  \node[grouplbl=pri, anchor=north west]
        at ([xshift=2pt]gPrep.north west) {Preprocessing};

  \node[groupbox=ter, fit=(offline)(online),
        inner xsep=8pt, inner ysep=8pt] (gAug) {};
  \node[grouplbl=ter, anchor=north west]
        at ([xshift=2pt]gAug.north west) {Augmentation Phase};
\end{scope}

\end{tikzpicture}%
}
