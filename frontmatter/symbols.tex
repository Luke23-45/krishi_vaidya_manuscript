\PrelimHeading{List of Symbols}
\PrelimEntry{List of Symbols}

\begin{longtable}{@{}p{0.22\textwidth}p{0.74\textwidth}@{}}
$P$ & Precision \\
$R$ & Recall \\
$F_1$ & F1 score \\
$IoU$ & Intersection over Union between predicted and ground-truth bounding boxes \\
$mAP_{50}$ & Mean average precision at IoU threshold 0.50 \\
$mAP_{50:95}$ & Mean average precision averaged across IoU thresholds from 0.50 to 0.95 \\
$TP$ & True positives \\
$FP$ & False positives \\
$FN$ & False negatives \\
$n$ & Number of samples, records, or observations \\
\end{longtable}
