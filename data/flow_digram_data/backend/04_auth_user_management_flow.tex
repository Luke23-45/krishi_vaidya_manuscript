                   minimum width=3.2cm, minimum height=1cm,

                    text width=3.8cm, minimum height=2cm,
                    fill=#1!5, draw=#1!22},
  arrlbl/.style={font=\sffamily\tiny\itshape, text=muted,
                 fill=white, inner sep=1.5pt, rounded corners=1pt},
  note/.style={rounded corners=7pt, draw=ink!10, fill=ink!2,

\definecolor{pri}{HTML}{2563EB}
\definecolor{sec}{HTML}{059669}
\definecolor{ter}{HTML}{D97706}
\definecolor{quat}{HTML}{7C3AED}
\definecolor{alert}{HTML}{DC2626}
\definecolor{bg}{HTML}{F8FAFC}
\definecolor{ink}{HTML}{0F172A}
\definecolor{muted}{HTML}{64748B}


% ━━ ROW 1: IDENTITY ESTABLISHMENT ━━
\node[statenode=sec] (reg) at (-5.8,-2.8) {
  \textbf{\textcolor{sec}{Registration}}\\[2pt]
    \textcolor{ink!72}{Validate identity fields\\}
  \textcolor{ink!72}{Hash password (bcrypt)\\}
  \textcolor{ink!72}{Create User document\\}
  \textcolor{ink!72}{Optional OTP email}};
\node[statenode=pri] (login) at (0,-2.8) {
  \textbf{\textcolor{pri}{Login}}\\[2pt]
    \textcolor{ink!72}{Verify email + password\\}
  \textcolor{ink!72}{Issue access token (JWT)\\}
  \textcolor{ink!72}{Store refresh token\\}
  \textcolor{ink!72}{Return user profile}};
\node[statenode=quat] (prot) at (5.8,-2.8) {
  \textbf{\textcolor{quat}{Protected Request}}\\[2pt]
    \textcolor{ink!72}{Extract Bearer token\\}
  \textcolor{ink!72}{Decode \& verify JWT\\}
  \textcolor{ink!72}{Attach \texttt{req.user}}\\
  \textcolor{ink!72}{Proceed to controller}};
\draw[flow] (reg) -- (login)
  node[arrlbl, pos=0.5, above, yshift=5pt] {credentials};
\draw[flow] (login) -- (prot)
  node[arrlbl, pos=0.5, above, yshift=5pt] {JWT issued};

% ━━ ROW 2: TOKEN LIFECYCLE ━━
\node[statenode=ter] (lifecycle) at (-5.8,-6.6) {
  \textbf{\textcolor{ter}{Token Lifecycle}}\\[2pt]
    \textcolor{ink!72}{Short-lived access token\\}
  \textcolor{ink!72}{Refresh token rotation\\}
  \textcolor{ink!72}{Logout single session\\}
  \textcolor{ink!72}{Logout all sessions\\}
  \textcolor{ink!72}{Invalidate on password change}};
\node[statenode=pri] (recovery) at (0,-6.6) {
  \textbf{\textcolor{pri}{Password Recovery}}\\[2pt]
    \textcolor{ink!72}{Send OTP to email\\}
  \textcolor{ink!72}{Verify OTP code\\}
  \textcolor{ink!72}{Reset password\\}
  \textcolor{ink!72}{Invalidate stale tokens\\}
  \textcolor{ink!72}{Rate-limited attempts}};
\node[statenode=alert] (outcome) at (5.8,-6.6) {
  \textbf{\textcolor{alert}{Auth Outcome}}\\[2pt]
    \textcolor{ink!72}{\textcolor{sec}{200} authorized controller call\\}
  \textcolor{ink!72}{\textcolor{alert}{401} missing/expired token\\}
  \textcolor{ink!72}{\textcolor{alert}{403} invalid ownership\\}
  \textcolor{ink!72}{\textcolor{ter}{429} rate limit exceeded\\}
  \textcolor{ink!72}{Global error handler}};
\draw[flow] (login.south) -- (recovery.north)
  node[arrlbl, midway, right, xshift=4pt] {forgot password};
\draw[flow] (login.south west) -- (lifecycle.north east)
  node[arrlbl, pos=0.42, left, xshift=-3pt, yshift=3pt] {token ops};
\draw[flow] (prot.south) -- (outcome.north)
  node[arrlbl, midway, right, xshift=5pt] {middleware result};

\draw[flow] (lifecycle) -- (recovery)
  node[arrlbl, pos=0.5, above, yshift=5pt] {token reset};
\draw[flow] (recovery) -- (outcome)
  node[arrlbl, pos=0.5, above, yshift=5pt] {re-auth};

% ━━ ROW 3: MIDDLEWARE DECISION ━━
\node[procnode=pri, minimum width=6cm, minimum height=1.2cm]
  (mw) at (0,-9.4) {
  \textbf{authMiddleware}\\[-1pt]
  {\tiny Decode JWT \textrightarrow{} verify expiry
   \textrightarrow{} lookup user
   \textrightarrow{} attach \texttt{req.user}
   \textrightarrow{} \texttt{next()}}};

\node[procnode=sec, minimum width=3cm] (ok) at (-4,-11.0) {
  \textbf{Authorized}\\[-1pt]{\tiny controller proceeds}};
\node[procnode=alert, minimum width=3cm] (deny) at (4,-11.0) {
  \textbf{Denied}\\[-1pt]{\tiny 401\,/\,403 JSON error}};

\draw[flow] (outcome.south) -- ++(0,-0.35) -| (mw.north);
\draw[flow] (mw.south west) -- (ok.north)
  node[arrlbl, pos=0.4, left] {valid};
\draw[errflow] (mw.south east) -- (deny.north)
  node[arrlbl, pos=0.4, right, text=alert!60] {invalid};

% ━━ NOTE ━━
\node[note] at (0,-12.6) {
  \textbf{Security}\;\textcolor{muted}{\textemdash}\;
  Session state is explicit: refresh tokens can be
  individually revoked without invalidating the
  stateless access-token verification path.
  The OTP recovery flow is rate-limited and
  time-bounded to prevent brute-force attacks.
  All sensitive operations log an audit trail.};

  \node[groupbox=sec, fit=(reg)(login)(prot),
        inner xsep=8pt, inner ysep=8pt] (gId) {};

        at ([xshift=2pt]gId.north west) {Identity Establishment};

  \node[groupbox=ter, fit=(lifecycle)(recovery)(outcome),
        inner xsep=8pt, inner ysep=8pt] (gLC) {};
  \node[grouplbl=ter, anchor=north west]
        at ([xshift=2pt]gLC.north west) {Token \& Recovery Lifecycle};

  \node[groupbox=pri, fit=(mw)(ok)(deny),
        inner xsep=8pt, inner ysep=8pt] (gMW) {};
  \node[grouplbl=pri, anchor=north west]
        at ([xshift=2pt]gMW.north west) {Middleware Decision};
\end{scope}
