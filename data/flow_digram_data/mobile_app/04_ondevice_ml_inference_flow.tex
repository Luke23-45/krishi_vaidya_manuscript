\begin{tikzpicture}[
  font=\sffamily,
  flow/.style={-{Latex[length=2.8mm,width=2mm]}, line width=0.8pt,
               draw=pri!70, rounded corners=6pt},
  flowret/.style={-{Latex[length=2.2mm,width=1.5mm]}, line width=0.5pt,
                  draw=muted!45, densely dashed},
  card/.style={rounded corners=10pt, draw=ink!10, line width=0.55pt,

  groupbox/.style={rounded corners=11pt, draw=#1!22, line width=0.4pt,
                   fill=#1!3},
  grouplbl/.style={font=\sffamily\bfseries\tiny, text=#1!70,
                   fill=white, inner xsep=3pt, inner ysep=1.5pt,
                   rounded corners=2pt},
  procnode/.style={card, align=center, inner sep=7pt,
                   minimum width=3.4cm, minimum height=1.1cm,
                   fill=#1!4, draw=#1!25},
  widenode/.style={card, align=center, inner sep=7pt,
                   minimum width=7.2cm, minimum height=1cm,
                   fill=#1!4, draw=#1!25},
  arrlbl/.style={font=\sffamily\tiny\itshape, text=muted,
                 fill=white, inner sep=1.5pt, rounded corners=1pt},
  note/.style={rounded corners=7pt, draw=ink!10, fill=ink!2,
               inner sep=8pt, align=left, font=\sffamily\tiny,

\definecolor{pri}{HTML}{E11D48}   % Rose
\definecolor{sec}{HTML}{0891B2}   % Cyan
\definecolor{ter}{HTML}{D97706}   % Amber
\definecolor{quat}{HTML}{7C3AED}  % Purple
\definecolor{bg}{HTML}{F8FAFC}
\definecolor{ink}{HTML}{0F172A}
\definecolor{muted}{HTML}{64748B}


% ━━ ROW 1: FRAME SOURCING ━━
\node[procnode=sec] (camera) at (0,-2.5) {
  \textbf{Frame Processor}\\[-1pt]
  {\tiny 60 FPS Camera Feed}};

% ━━ ROW 2: PREPROCESSING (JSI) ━━
\node[widenode=ter] (jsi) at (0,-4.5) {
  \textbf{C++ / JSI Bridge (No-Copy)}\\[2pt]
    \textcolor{ink!72}{\tiny $\bullet$ YUV to RGB Conversion}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ Resize to $640\times 640$}\\[-1pt]
  \textcolor{ink!72}{{\tiny $\bullet$ Normalize $(0-255 \rightarrow 0-1)$}}};
\draw[flow] (camera) -- (jsi) node[arrlbl, midway, right] {HostObject};

% ━━ ROW 3: INTERPRETER ━━
\node[widenode=pri] (tflite) at (0,-6.7) {
  \textbf{TFLite Runtime}\\[2pt]
    \textcolor{ink!72}{\tiny $\bullet$ Hardware Delegate (NPU/GPU/NNAPI)}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ Execute INT8 YOLO Checkpoint}\\[-1pt]
  \textcolor{ink!72}{{\tiny $\bullet$ Output Tensor: Bounding boxes \& confidences}}};
\draw[flow] (jsi) -- (tflite) node[arrlbl, midway, right] {Tensor Float32/Int8};

% ━━ ROW 4: POST & RENDER ━━
\node[procnode=sec] (nms) at (-3.5,-9.0) {
  \textbf{NMS Filtering}\\[-1pt]
  {\tiny Remove overlaps}};

\node[procnode=quat] (render) at (3.5,-9.0) {
  \textbf{Skia Render}\\[-1pt]
  {\tiny Draw boxes on view}};

\draw[flow] (tflite.south west) -- ++(0,-0.4) -| (nms.north);
\draw[flow] (nms.east) -- (render.west);

% ━━ NOTE ━━
\node[note] at (0,-11.0) {
  \textbf{Performance}\;\textcolor{muted}{\textemdash}\;
  To achieve real-time (30+ FPS) AR overlays, frames are never serialized
  to base64 or passed over the React Native bridge. Instead, C++ JSI bindings
  manage the memory pointers directly from the camera buffer to the TFLite runtime,
  preventing garbage collection stutters.};

% ━━ BACKGROUND ━━
\begin{scope}[on background layer]
  \node[rounded corners=14pt, fill=bg!45, draw=ink!4,
        line width=0.35pt, inner sep=14pt,
        fit=(camera)(jsi)(tflite)(nms)(render)] {};

  \node[groupbox=ter, fit=(jsi)(tflite),
        inner xsep=10pt, inner ysep=8pt] (gNative) {};
  \node[grouplbl=ter, anchor=north west]
        at ([xshift=2pt]gNative.north west) {Native Layer (C++)};
\end{scope}

\end{tikzpicture}%
