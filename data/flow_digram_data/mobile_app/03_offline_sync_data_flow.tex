\resizebox{0.98\textwidth}{!}{%
\begin{tikzpicture}[
  font=\sffamily,
  flow/.style={-{Latex[length=2.8mm,width=2mm]}, line width=0.8pt,
               draw=pri!70, rounded corners=6pt},
  flowret/.style={-{Latex[length=2.2mm,width=1.5mm]}, line width=0.5pt,
                  draw=muted!45, densely dashed},
  card/.style={rounded corners=10pt, draw=ink!10, line width=0.55pt,
               fill=white, align=left, inner sep=8pt,
               font=\sffamily\scriptsize, text=ink,
               blur shadow={shadow blur steps=5, shadow xshift=0.6pt,
               shadow yshift=-1pt, shadow opacity=13}},
  groupbox/.style={rounded corners=11pt, draw=#1!22, line width=0.4pt,
                   fill=#1!3},
  grouplbl/.style={font=\sffamily\bfseries\tiny, text=#1!70,
                   fill=white, inner xsep=3pt, inner ysep=1.5pt,
                   rounded corners=2pt},
  titlebar/.style={rounded corners=8pt, fill=pri, text=white,
                   inner xsep=14pt, inner ysep=7pt,
                   font=\sffamily\bfseries\large},
  procnode/.style={card, align=center, inner sep=7pt,
                   minimum width=3.4cm, minimum height=1.1cm,
                   fill=#1!4, draw=#1!25},
  dbnode/.style={card, align=center, inner sep=7pt,
                 minimum width=3.4cm, minimum height=1.1cm,
                 fill=ter!5, draw=ter!25},
  arrlbl/.style={font=\sffamily\tiny\itshape, text=muted,
                 fill=white, inner sep=1.5pt, rounded corners=1pt},
  note/.style={rounded corners=7pt, draw=ink!10, fill=ink!2,
               inner sep=8pt, align=left, font=\sffamily\tiny,
               text=ink!78, text width=14.2cm}
]

\definecolor{pri}{HTML}{E11D48}   % Rose
\definecolor{sec}{HTML}{0891B2}   % Cyan
\definecolor{ter}{HTML}{D97706}   % Amber
\definecolor{quat}{HTML}{7C3AED}  % Purple
\definecolor{bg}{HTML}{F8FAFC}
\definecolor{ink}{HTML}{0F172A}
\definecolor{muted}{HTML}{64748B}

% ━━ TITLE ━━
\node[titlebar, minimum width=15.2cm] at (0,0.3)
  {Mobile Offline Sync Data Flow};
\node[font=\sffamily\footnotesize, text=muted, text width=14.8cm,
      align=center] at (0,-1.0)
  {Handling user actions without internet connectivity using the local
   SQLite store and queued sync operations, then flushing them when
   network connectivity returns.};

% ━━ ROW 1: USER ACTION (Offline) ━━
\node[procnode=sec] (action) at (-4,-2.5) {
  \textbf{User Action}\\[-1pt]
  {\tiny e.g., Save Task, Add Note}};

\node[dbnode] (local) at (4,-2.5) {
  \textbf{Local SQLite Store}\\[-1pt]
  {\tiny expo-sqlite + sync\_queue tables}};

\draw[flow] (action) -- (local) node[arrlbl, midway, above] {write with \texttt{sync\_status='pending'}};

% ━━ ROW 2: NETWORK LISTENER ━━
\node[procnode=pri, minimum width=6cm] (listener) at (0,-4.8) {
  \textbf{Network Connectivity Listener}\\[-1pt]
  {\tiny React Native NetInfo detects \texttt{isConnected == true}}};

% ━━ ROW 3: SYNC ENGINE ━━
\node[procnode=quat, minimum width=6cm] (sync) at (0,-6.8) {
  \textbf{Background Sync Engine}\\[2pt]
    \textcolor{ink!72}{\tiny $\bullet$ Query all \texttt{sync\_status='pending'}}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ Batch POST to Backend API}\\[-1pt]
  \textcolor{ink!72}{{\tiny $\bullet$ Conflict resolution (timestamp)}}};
\draw[flow] (listener) -- (sync) node[arrlbl, midway, right] {trigger};
\draw[flow] (local.south) |- (sync.east) node[arrlbl, pos=0.75, below] {read pending};

% ━━ ROW 4: COMPLETION ━━
\node[dbnode] (cloud) at (-4,-8.8) {
  \textbf{Backend API}\\[-1pt]
  {\tiny Express / REST}};

\draw[flow] (sync.south) |- (cloud.east) node[arrlbl, pos=0.8, above] {HTTP 200};
\draw[flowret] (cloud.north) -- (action.south) node[arrlbl, midway, left] {Update UI (Synced)};

% ━━ NOTE ━━
\node[note] at (0,-10.8) {
  \textbf{Offline-First}\;\textcolor{muted}{\textemdash}\;
  The UI always reads from and writes to the local SQLite-backed data layer, ensuring
  zero latency and full offline support. The Sync Engine runs as a background job
  that pushes local mutations to the cloud and pulls fresh data down when the
  network is available.};

% ━━ BACKGROUND ━━
\begin{scope}[on background layer]
  \node[rounded corners=14pt, fill=bg!45, draw=ink!4,
        line width=0.35pt, inner sep=14pt,
        fit=(action)(local)(listener)(sync)(cloud)] {};

  \node[groupbox=ter, fit=(local)(sync),
        inner xsep=8pt, inner ysep=8pt] (gLocal) {};
  \node[grouplbl=ter, anchor=north east]
        at ([xshift=-2pt]gLocal.north east) {Local Storage \& Queue};
\end{scope}

\end{tikzpicture}%
}
