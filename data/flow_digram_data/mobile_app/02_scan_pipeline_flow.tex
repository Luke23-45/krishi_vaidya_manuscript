\begin{tikzpicture}[
  font=\sffamily,
  flow/.style={-{Latex[length=2.8mm,width=2mm]}, line width=0.8pt,
               draw=pri!70, rounded corners=6pt},
  flowret/.style={-{Latex[length=2.2mm,width=1.5mm]}, line width=0.5pt,
                  draw=muted!45, densely dashed},
  card/.style={rounded corners=10pt, draw=ink!10, line width=0.55pt,
               fill=white, align=left, inner sep=8pt,
               font=\sffamily\scriptsize, text=ink,
               blur shadow={shadow blur steps=5, shadow xshift=0.6pt,
               shadow yshift=-1pt, shadow opacity=13}},
  groupbox/.style={rounded corners=11pt, draw=#1!22, line width=0.4pt,
                   fill=#1!3},
  grouplbl/.style={font=\sffamily\bfseries\tiny, text=#1!70,
                   fill=white, inner xsep=3pt, inner ysep=1.5pt,
                   rounded corners=2pt},
  procnode/.style={card, align=center, inner sep=7pt,
                   minimum width=3.4cm, minimum height=1.1cm,
                   fill=#1!4, draw=#1!25},
  widenode/.style={card, align=center, inner sep=7pt,
                   minimum width=8cm, minimum height=1cm,
                   fill=#1!4, draw=#1!25},
  arrlbl/.style={font=\sffamily\tiny\itshape, text=muted,
                 fill=white, inner sep=1.5pt, rounded corners=1pt},
  note/.style={rounded corners=7pt, draw=ink!10, fill=ink!2,
               inner sep=8pt, align=left, font=\sffamily\tiny,
               text=ink!78, text width=14.2cm}
]

\definecolor{pri}{HTML}{E11D48}   % Rose
\definecolor{sec}{HTML}{0891B2}   % Cyan
\definecolor{ter}{HTML}{D97706}   % Amber
\definecolor{quat}{HTML}{7C3AED}  % Purple
\definecolor{alert}{HTML}{DC2626}
\definecolor{bg}{HTML}{F8FAFC}
\definecolor{ink}{HTML}{0F172A}
\definecolor{muted}{HTML}{64748B}


% ━━ ROW 1: CAPTURE ━━
\node[procnode=sec] (camera) at (0,-2.3) {
  \textbf{Camera View}\\[-1pt]
  {\tiny React Native Vision Camera}};

\node[procnode=ter] (capture) at (0,-4.1) {
  \textbf{Image Capture}\\[-1pt]
  {\tiny Resize to 1024px, compress JPEG}};

\draw[flow] (camera) -- (capture) node[arrlbl, midway, right] {user triggers shutter};

% ━━ ROW 2: NETWORK ORCHESTRATION ━━
\node[widenode=pri] (api) at (0,-6.3) {
  \textbf{API Orchestrator}\\[2pt]
    \textcolor{ink!72}{\tiny $\bullet$ Show Loading UI Overlay}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ Attach JWT Header \& Multi-part Form Data}\\[-1pt]
  \textcolor{ink!72}{{\tiny $\bullet$ POST /diagnose (Timeout: 15s)}}};
\draw[flow] (capture) -- (api) node[arrlbl, midway, right] {base64 string};

% ━━ ROW 3: OUTCOMES ━━
\node[procnode=quat] (success) at (-3.5,-8.8) {
  \textbf{Results Screen}\\[-1pt]
  {\tiny Display Disease \& Remedies}};

\node[procnode=muted, draw=alert!50] (fail) at (3.5,-8.8) {
  \textbf{Error Handler}\\[-1pt]
  {\tiny Toast: "Network Error" / Retry}};

\draw[flow] (api.south west) -- ++(0,-0.4) -| (success.north) node[arrlbl, pos=0.7, left] {HTTP 200};
\draw[flowret, draw=alert!60] (api.south east) -- ++(0,-0.4) -| (fail.north) node[arrlbl, pos=0.7, right, text=alert!80] {Timeout / 5xx};

% ━━ NOTE ━━
\node[note] at (0,-10.8) {
  \textbf{UX Considerations}\;\textcolor{muted}{\textemdash}\;
  Image compression happens natively on the thread to prevent UI freezing.
  During the API call, a custom skeleton loader or Lottie animation is
  shown to keep the user engaged while waiting for the VLM backend inference.};

% ━━ BACKGROUND ━━
\begin{scope}[on background layer]
  \node[rounded corners=14pt, fill=bg!45, draw=ink!4,
        line width=0.35pt, inner sep=14pt,
        fit=(camera)(capture)(api)(success)(fail)] {};

  \node[groupbox=sec, fit=(camera)(capture),
        inner xsep=10pt, inner ysep=8pt] (gCap) {};
  \node[grouplbl=sec, anchor=north west]
        at ([xshift=2pt]gCap.north west) {Local Device};
\end{scope}

\end{tikzpicture}%
