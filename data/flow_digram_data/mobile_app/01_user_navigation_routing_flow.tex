  {React Navigation state machine: Switch from Splash/Auth stack to the

% ━━ ROW 1: ENTRY POINT ━━
\node[scrcard=pri] (splash) at (0,-2.3) {
  \textbf{Splash Screen}\\[-1pt]
  {\tiny Check Async Storage}};

% ━━ ROW 2: AUTH STACK (Left) vs APP STACK (Right) ━━
\node[scrcard=ter] (auth) at (-4,-4.3) {
  \textbf{Auth Stack}\\[-1pt]
  {\tiny Login / Register}};

\node[scrcard=sec] (home) at (4,-4.3) {
  \textbf{Main App (Tabs)}\\[-1pt]
  {\tiny Dashboard, History, Profile}};

\draw[flow] (splash) -- (auth) node[arrlbl, midway, left] {no token};
\draw[flow] (splash) -- (home) node[arrlbl, midway, right] {valid JWT};
\draw[flow] (auth) -- (home) node[arrlbl, midway, above] {success};

% ━━ ROW 3: NESTED SCAN FLOW ━━
\node[scrcard=quat] (scan) at (4,-7) {
  \textbf{Scan Modal Stack}\\[2pt]
    \textcolor{ink!72}{\tiny 1. Camera View}\\[-1pt]
  \textcolor{ink!72}{\tiny 2. Crop Select}\\[-1pt]
  \textcolor{ink!72}{{\tiny 3. Results View}}};
\draw[flow] (home) -- (scan) node[arrlbl, midway, right] {FAB press};
\draw[flowret] (scan.160) -- (home.200) node[arrlbl, midway, left] {dismiss};

% ━━ ROW 4: LOGOUT ━━
\draw[flowret, draw=alert!50]
  (home.west) -- ++(-0.55,0) |- ([yshift=-0.45cm]auth.south east) -- (auth.south east)
  node[arrlbl, pos=0.62, below] {logout / expire};

% ━━ NOTE ━━
\node[note] at (0,-9.2) {
  \textbf{Navigation Hierarchy}\;\textcolor{muted}{\textemdash}\;

  and the Main App Stack based on the JWT state, ensuring unauthenticated users
  cannot deep-link into protected tabs. The Scan Stack is presented modally over
  the bottom tabs to preserve context.};
