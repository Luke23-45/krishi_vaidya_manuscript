% ━━ ROW 1: INPUT ━━
\node[procnode=pri] (testset) at (0,-2.2) {
  \textbf{Holdout Test Set}\\[-1pt]
  {\tiny Ground truth annotations}};

% ━━ ROW 2: FORK ━━
\node[procnode=sec] (gen) at (-3.5,-4.2) {
  \textbf{Model Generation}\\[2pt]
    \textcolor{ink!72}{\tiny $\bullet$ Feed images to vLLM}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ Enforce JSON schema}\\[-1pt]
  \textcolor{ink!72}{{\tiny $\bullet$ Extract disease ID}}};
\node[procnode=ter] (metric) at (3.5,-4.2) {
  \textbf{Metrics Calculation}\\[2pt]
    \textcolor{ink!72}{\tiny $\bullet$ Accuracy / Precision / Recall}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ Per-class breakdown}\\[-1pt]
  \textcolor{ink!72}{{\tiny $\bullet$ Confusion matrix}}};
\draw[flow] (testset.south west) -- ++(0,-0.3) -| (gen.north);
\draw[flow] (testset.south east) -- ++(0,-0.3) -| (metric.north) node[arrlbl, pos=0.8, right] {truth};

\draw[flow] (gen.east) -- (metric.west) node[arrlbl, midway, above] {predictions};

% ━━ ROW 3: JOIN / OUTPUTS ━━
\node[resultcard, text width=3.8cm] (report) at (-2.5,-6.8) {
  \textbf{Eval Report}\\[-1pt]
  {\tiny JSON metrics, Markdown summary}};

\node[resultcard, text width=3.8cm] (plots) at (2.5,-6.8) {
  \textbf{Visualizations}\\[-1pt]
  {\tiny Confusion Matrix (PNG)}};

\draw[flow] (metric.south) -- ++(0,-0.4) -| (report.north);
\draw[flow] (metric.south) -- ++(0,-0.4) -| (plots.north);

% ━━ NOTE ━━
\node[note] at (0,-8.8) {
  \textbf{Process}\;\textcolor{muted}{\textemdash}\;
  Validation ensures the LoRA weights actually learned the agricultural domain
  by testing against a pristine holdout set. Because generation is constrained to JSON,
  predictions can be programmatically compared to ground truth without human grading.};
