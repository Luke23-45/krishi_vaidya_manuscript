\definecolor{pri}{HTML}{4338CA}   % Indigo
\definecolor{sec}{HTML}{0D9488}   % Teal
\definecolor{ter}{HTML}{D97706}   % Amber
\definecolor{quat}{HTML}{9333EA}  % Purple
\definecolor{bg}{HTML}{F8FAFC}
\definecolor{ink}{HTML}{0F172A}
\definecolor{muted}{HTML}{64748B}


% ━━ ROW 1: INPUTS (Funnel) ━━
\node[procnode=sec] (ds) at (-3.5,-2.2) {
  \textbf{Processed Dataset}\\[-1pt]
  {\tiny HuggingFace (from Stage 1)}};
\node[procnode=ter] (model) at (3.5,-2.2) {
  \textbf{Base Model}\\[-1pt]
  {\tiny Qwen/Qwen2.5-VL-3B-Instruct}};

% ━━ ROW 2: PREPARATION ━━
\node[procnode=pri, text width=5cm] (prep) at (0,-3.8) {
  \textbf{Preparation}\\[-1pt]
  {\tiny Load model in bfloat16\\[-1pt]
   Initialize Qwen2.5\_VLProcessor}};

\draw[flow] (ds.south) -- ++(0,-0.3) -| ([xshift=-15pt]prep.north);
\draw[flow] (model.south) -- ++(0,-0.3) -| ([xshift=15pt]prep.north);

% ━━ ROW 3: PEFT ━━
\node[procnode=quat, text width=6cm] (peft) at (0,-5.6) {
  \textbf{PEFT Configuration (LoRA)}\\[-1pt]
  {\tiny rank $r=16$, $\alpha=32$\\[-1pt]
   target: q\_proj, k\_proj, v\_proj, o\_proj}};

\draw[flow] (prep) -- (peft);

% ━━ ROW 4: ACCELERATOR TRAINER ━━
\node[procnode=pri, text width=7.2cm, minimum height=2.2cm] (trainer) at (0,-8.0) {
  \textbf{SFTTrainer (TRL)}\\[2pt]
    \textcolor{ink!72}{\tiny $\bullet$ HF Accelerate (Distributed DDP/DeepSpeed)}\\[-1pt]
  \textcolor{ink!72}{\tiny $\bullet$ Optimizer: AdamW, Scheduler: Cosine}\\[-1pt]
  \textcolor{ink!72}{{\tiny $\bullet$ Logging: Weights \& Biases}}};
\draw[flow] (peft) -- (trainer);

% ━━ ROW 5: OUTPUTS ━━
\node[procnode=sec] (adapter) at (-3.5,-10.8) {
  \textbf{LoRA Adapters}\\[-1pt]
  {\tiny adapter\_model.safetensors}};
\node[procnode=ter] (merged) at (3.5,-10.8) {
  \textbf{Merged Model}\\[-1pt]
  {\tiny final full weights}};

\draw[flow] (trainer.south west) -- ++(0,-0.4) -| (adapter.north);
\draw[flow] (trainer.south east) -- ++(0,-0.4) -| (merged.north)
  node[arrlbl, pos=0.8, right] {optional};

% ━━ NOTE ━━
\node[note] at (0,-12.6) {
  \textbf{Architecture}\;\textcolor{muted}{\textemdash}\;
  Training leverages parameter-efficient fine-tuning (PEFT) with LoRA
  to adapt the vision and language projection layers without modifying
  the massive base model. Bfloat16 precision prevents underflow issues
  common in visual token scaling.};

  \node[groupbox=pri, fit=(prep)(peft)(trainer),
        inner xsep=8pt, inner ysep=8pt] (gTrain) {};
  \node[grouplbl=pri, anchor=north west]
        at ([xshift=6pt,yshift=-5pt]gTrain.north west) {Training Loop};
\end{scope}
