% ━━ TITLE ━━

  {VLM Data Pipeline Flow (11 Stages)};
\node[font=\sffamily\footnotesize, text=muted, text width=14cm,
      align=center] at (0,-1.0)
  {The serpentine sequence of ingestion, text cleaning, visual curation,
   and formatting into Qwen-style conversational arrays.};

% ━━ ROW 1: INGESTION (Right to Left to start serpentine, or Left to Right) ━━
% Let's do Left to Right for Row 1
\node[stagenode=pri] (s1) at (-6.0,-2.5) {
  \textbf{1. Raw Ingestion}\\[-1pt]
  {\tiny JSONL\,/\,CSV sources}};
\node[stagenode=pri] (s2) at (-2.0,-2.5) {
  \textbf{2. Deduplication}\\[-1pt]
  {\tiny URL\,/\,Hash collision}};
\node[stagenode=pri] (s3) at ( 2.0,-2.5) {
  \textbf{3. Text Cleaning}\\[-1pt]
  {\tiny regex, markdown}};
\node[stagenode=pri] (s4) at ( 6.0,-2.5) {
  \textbf{4. Token Truncation}\\[-1pt]
  {\tiny enforce max length}};

\draw[flow] (s1) -- (s2);
\draw[flow] (s2) -- (s3);
\draw[flow] (s3) -- (s4);

% ━━ ROW 2: VISION VALIDATION (Right to Left) ━━
\node[stagenode=sec] (s5) at ( 6.0,-5.0) {
  \textbf{5. Image Download}\\[-1pt]
  {\tiny fetch URLs, timeout}};
\node[stagenode=sec] (s6) at ( 2.0,-5.0) {
  \textbf{6. Format Verify}\\[-1pt]
  {\tiny JPEG/PNG headers}};
\node[stagenode=sec] (s7) at (-2.0,-5.0) {
  \textbf{7. Visual Curation}\\[-1pt]
  {\tiny blur, resolution}};
\node[stagenode=sec] (s8) at (-6.0,-5.0) {
  \textbf{8. Resize \& Pad}\\[-1pt]
  {\tiny conform to 1024px}};

% serpentine curve down
\draw[flow] (s4.east) -- ++(0.6,0) |- (s5.east);

\draw[flow] (s5) -- (s6);
\draw[flow] (s6) -- (s7);
\draw[flow] (s7) -- (s8);

% ━━ ROW 3: EXPORT (Left to Right) ━━
\node[stagenode=ter] (s9) at (-6.0,-7.5) {
  \textbf{9. Structuring}\\[-1pt]
  {\tiny map to conversation}};
\node[stagenode=ter] (s10) at (-2.0,-7.5) {
  \textbf{10. Tokenization}\\[-1pt]
  {\tiny Qwen2.5 chat template}};
\node[stagenode=ter] (s11) at ( 2.0,-7.5) {
  \textbf{11. Packing}\\[-1pt]
  {\tiny batch chunks}};
\node[stagenode=quat, minimum width=2.8cm, fill=quat!8, draw=quat!35]
  (out) at ( 6.0,-7.5) {
  \textbf{Final Dataset}\\[-1pt]
  {\tiny HuggingFace format}};

% serpentine curve down
\draw[flow] (s8.west) -- ++(-0.6,0) |- (s9.west);

\draw[flow] (s9) -- (s10);
\draw[flow] (s10) -- (s11);
\draw[flow] (s11) -- (out) node[arrlbl, midway, above] {dataset export};

% ━━ DROPPED SAMPLES (vertical cascades) ━━
\node[font=\sffamily\tiny\itshape, text=alert] (drop1) at (2.0,-3.7) {drop invalid text};
\draw[flowret, draw=alert!60] (s3.south) -- (drop1.north);

\node[font=\sffamily\tiny\itshape, text=alert] (drop2) at (-2.0,-6.2) {drop corrupt/blur};
\draw[flowret, draw=alert!60] (s7.south) -- (drop2.north);

% ━━ NOTE ━━
\node[note] at (0,-9.2) {
  \textbf{Design rationale}\;\textcolor{muted}{\textemdash}\;
  The pipeline is strictly sequential. Text cleaning occurs before image downloading
  to save bandwidth by immediately dropping malformed textual records. Visual
  curation guarantees that the model receives only crisp, adequately sized imagery,
  which is critical for the visual token representations in the Qwen architecture.};
